$projectPath = "D:\barnamenevisi\uidesigns\tradehub"
$outputFile  = Join-Path $projectPath "project-source.txt"

$excludeDirs = @(
    "node_modules", ".next", ".git", "dist", "build",
    ".vercel", ".idea", ".vscode", ".agents", ".claude"
)

$excludeExts = @(
    ".png", ".jpg", ".jpeg", ".gif", ".ico", ".svg", ".webp",
    ".pdf", ".zip", ".rar", ".7z", ".exe", ".dll", ".lock",
    ".log", ".woff", ".woff2", ".ttf", ".eot", ".mp4", ".mp3",
    ".xls", ".xlsx"
)

$excludeFiles = @("package-lock.json", "project-source.txt")

# پاک کردن فایل خروجی قبلی
if (Test-Path -LiteralPath $outputFile) {
    Remove-Item -LiteralPath $outputFile -Force
}

# ساخت الگوی رجولار امن برای پوشه‌ها
$escapedDirs = $excludeDirs | ForEach-Object { [regex]::Escape($_) }
$dirPattern  = "\\($($escapedDirs -join '|'))\\"

# دریافت فایل‌ها
$files = Get-ChildItem -LiteralPath $projectPath -Recurse -File -ErrorAction SilentlyContinue |
    Where-Object {
        $dirMatch  = $_.FullName -match $dirPattern
        $extMatch  = $excludeExts -contains $_.Extension.ToLower()
        $fileMatch = $excludeFiles -contains $_.Name
        -not $dirMatch -and -not $extMatch -and -not $fileMatch
    }

# نوشتن محتوا
foreach ($file in $files) {
    $relativePath = $file.FullName.Substring($projectPath.Length).TrimStart("\")

    Add-Content -LiteralPath $outputFile -Encoding UTF8 -Value "`n`n=========================================="
    Add-Content -LiteralPath $outputFile -Encoding UTF8 -Value "FILE: $relativePath"
    Add-Content -LiteralPath $outputFile -Encoding UTF8 -Value "==========================================`n"

    try {
        # -LiteralPath نکته کلیدی برای مسیرهای دارای [ ] است
        $content = Get-Content -LiteralPath $file.FullName -Raw -ErrorAction Stop
        Add-Content -LiteralPath $outputFile -Encoding UTF8 -Value $content
    }
    catch {
        Add-Content -LiteralPath $outputFile -Encoding UTF8 `
            -Value "[ERROR reading file: $($_.Exception.Message)]"
    }
}

Write-Host "Done. Output: $outputFile" -ForegroundColor Green
